def cargar_productos(conn):
    cur = conn.cursor(dictionary=True)
    cur.execute("""
        SELECT cod_producto, nombre, precio, categoria
        FROM productos
        WHERE activo = 1
        ORDER BY cod_producto
    """)
    productos = cur.fetchall()
    for p in productos:
        p["precio"] = float(p["precio"])

2) Reemplaza la función registrar_pedido por:

def registrar_pedido(conn, n_cliente, cantidades, metodo, direccion_entrega, productos):
    """Guarda la cabecera en pedidos y los productos en ventas."""
    cur = conn.cursor()
    total = sum(float(productos[cod]["precio"]) * cantidad
                for cod, cantidad in cantidades.items())
    conn.start_transaction()
    try:
        cur.execute(
            "INSERT INTO pedidos (id_comprador, id_vendedor, metodo_pago, direccion_entrega, total) "
            "VALUES (%s, %s, %s, %s, %s)",
            (n_cliente, ID_VENDEDOR_ONLINE, metodo, direccion_entrega, total),
        )
        n_pedido = cur.lastrowid
        cur.executemany(
            "INSERT INTO ventas (n_pedido, id_product, u_vendidas, precio_unitario) "
            "VALUES (%s, %s, %s, %s)",
            [(n_pedido, cod, cantidad, productos[cod]["precio"])
             for cod, cantidad in cantidades.items()],
        )
        conn.commit()
        return n_pedido
    except Exception:
        conn.rollback()
        raise
